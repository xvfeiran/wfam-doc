#### 4.2.2 SamplingModal（抽样模态框）

```typescript
// src/features/return-orders/components/SamplingModal.tsx
interface SamplingModalProps {
  orderId: string;
  parts: PartItem[];
  onConfirm: (samplingData: SamplingData) => void;
  onClose: () => void;
}

interface SamplingData {
  method: 'standard' | 'custom' | 'none';
  selectedParts: Array<{
    partId: string;
    sampleNumber: string;
    isRepresentative: boolean;
  }>;
}

export const SamplingModal: React.FC<SamplingModalProps> = ({
  orderId,
  parts,
  onConfirm,
  onClose,
}) => {
  const [method, setMethod] = useState<'standard' | 'custom' | 'none'>('standard');
  const [selectedParts, setSelectedParts] = useState<string[]>([]);

  const generateSampleNumber = (part: PartItem, index: number) => {
    return `${part.businessUnit}-${part.productPlatform}-${String(index + 1).padStart(4, '0')}`;
  };

  const handleConfirm = () => {
    const samplingData: SamplingData = {
      method,
      selectedParts: selectedParts.map((partId, index) => {
        const part = parts.find(p => p.id === partId)!;
        return {
          partId,
          sampleNumber: generateSampleNumber(part, index),
          isRepresentative: true,
        };
      }),
    };
    onConfirm(samplingData);
  };

  return (
    <Modal isOpen onClose={onClose} title="抽样管理" size="large">
      <div className="space-y-6">
        {/* 抽样方式选择 */}
        <div>
          <label className="block text-sm font-medium mb-2">抽样方式</label>
          <div className="flex gap-4">
            <Radio
              label="标准抽样"
              checked={method === 'standard'}
              onChange={() => setMethod('standard')}
            />
            <Radio
              label="指定抽样"
              checked={method === 'custom'}
              onChange={() => setMethod('custom')}
            />
            <Radio
              label="不抽样"
              checked={method === 'none'}
              onChange={() => setMethod('none')}
            />
          </div>
        </div>

        {method !== 'none' && (
          <div className="grid grid-cols-2 gap-4">
            {/* 左侧：可选零件列表 */}
            <div className="border rounded p-4">
              <h4 className="font-medium mb-2">可选零件</h4>
              <div className="space-y-2 max-h-96 overflow-y-auto">
                {parts.map((part) => (
                  <Checkbox
                    key={part.id}
                    label={`${part.partCode} - ${part.businessUnit}`}
                    checked={selectedParts.includes(part.id)}
                    onChange={(checked) => {

export interface OCRResult {
  field: 'productionDate' | 'purchaseDate' | 'failureDate' | 'vin' | 'mileage' | 'description';
  value: string;
  confidence: number;
  status: 'loading' | 'success' | 'error';

// src/features/return-parts/hooks/useOCR.ts
import { useState } from 'react';
import { mockOCRService } from '@/services/ocr.service';

export const useOCR = () => {
  const [isScanning, setIsScanning] = useState(false);
  const [results, setResults] = useState<Record<string, OCRResult>>({});

  const startScan = async (image: File) => {
    setIsScanning(true);

    // 设置所有字段为loading状态
    const fields: OCRResult['field'][] = ['productionDate', 'purchaseDate', 'failureDate', 'vin', 'mileage', 'description'];
    const loadingResults: Record<string, OCRResult> = {};
    fields.forEach(field => {
      loadingResults[field] = { field, value: '', confidence: 0, status: 'loading' };
    });
    setResults(loadingResults);

    try {
      // 调用OCR服务
      const scanResults = await mockOCRService.scanImage(image);

      // 更新结果
      const successResults: Record<string, OCRResult> = {};
      Object.entries(scanResults).forEach(([key, value]) => {
        successResults[key] = {
          field: key as OCRResult['field'],
          value: value as string,
          confidence: 0.95,
          status: 'success',
        };
      });
      setResults(successResults);
    } catch (error) {
      console.error('OCR扫描失败:', error);
      // 设置错误状态
      const errorResults: Record<string, OCRResult> = {};
      fields.forEach(field => {
        errorResults[field] = { field, value: '', confidence: 0, status: 'error' };
      });
      setResults(errorResults);
    } finally {
      setIsScanning(false);
    }
  };

  const stopScan = () => {
    setIsScanning(false);
    setResults({});
  };

  const applyResult = (field: string, onApply: (value: string) => void) => {
    const result = results[field];
    if (result && result.status === 'success') {
      onApply(result.value);
    }
  };

  const applyAllResults = (onApplyAll: (results: Record<string, string>) => void) => {
    const values: Record<string, string> = {};
    Object.entries(results).forEach(([field, result]) => {
      if (result.status === 'success') {
        values[field] = result.value;
      }
    });
    onApplyAll(values);
  };

  return {
    isScanning,
    results,
    startScan,
    stopScan,
    applyResult,
    applyAllResults,
  };
};

  orders: ReturnOrder[];
  currentOrder: ReturnOrder | null;
  filters: OrderFilters;
  isLoading: boolean;

  // Actions
  setOrders: (orders: ReturnOrder[]) => void;
  setCurrentOrder: (order: ReturnOrder | null) => void;
  updateFilters: (filters: Partial<OrderFilters>) => void;
  addOrder: (order: ReturnOrder) => void;
  updateOrder: (id: string, order: Partial<ReturnOrder>) => void;
  deleteOrder: (id: string) => void;
  setLoading: (loading: boolean) => void;
}

export const useOrderStore = create<OrderState>((set) => ({
  orders: [],
  currentOrder: null,
  filters: {},
  isLoading: false,

  setOrders: (orders) => set({ orders }),
  setCurrentOrder: (order) => set({ currentOrder: order }),
  updateFilters: (filters) => set((state) => ({ filters: { ...state.filters, ...filters } })),
  addOrder: (order) => set((state) => ({ orders: [...state.orders, order] })),
  updateOrder: (id, updatedOrder) => set((state) => ({
    orders: state.orders.map((order) =>
      order.id === id ? { ...order, ...updatedOrder } : order
    ),
  })),
  deleteOrder: (id) => set((state) => ({
    orders: state.orders.filter((order) => order.id !== id),
  })),
  setLoading: (loading) => set({ isLoading: loading }),
}));

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
  size?: 'sm' | 'md' | 'lg' | 'xl';
  children: React.ReactNode;
}

export const Modal: React.FC<ModalProps> = ({
  isOpen,
  onClose,
  title,
  size = 'md',
  children,
}) => {
  const sizeStyles = {
    sm: 'max-w-md',
    md: 'max-w-lg',
    lg: 'max-w-2xl',
    xl: 'max-w-4xl',
  };

  return (
    <Transition appear show={isOpen} as={Fragment}>
      <Dialog as="div" className="relative z-50" onClose={onClose}>
        <Transition.Child
          as={Fragment}
          enter="ease-out duration-300"
          enterFrom="opacity-0"
          enterTo="opacity-100"
          leave="ease-in duration-200"
          leaveFrom="opacity-100"
          leaveTo="opacity-0"
        >
          <div className="fixed inset-0 bg-black bg-opacity-25" />
        </Transition.Child>

        <div className="fixed inset-0 overflow-y-auto">
          <div className="flex min-h-full items-center justify-center p-4">
            <Transition.Child
              as={Fragment}
              enter="ease-out duration-300"
              enterFrom="opacity-0 scale-95"
              enterTo="opacity-100 scale-100"
              leave="ease-in duration-200"
              leaveFrom="opacity-100 scale-100"
              leaveTo="opacity-0 scale-95"
            >
              <Dialog.Panel className={cn('w-full transform overflow-hidden rounded-lg bg-white p-6 shadow-xl transition-all', sizeStyles[size])}>
                {title && (
                  <div className="flex items-center justify-between mb-4">
                    <Dialog.Title className="text-lg font-semibold">{title}</Dialog.Title>
                    <button onClick={onClose} className="text-gray-400 hover:text-gray-600">
                      <XMarkIcon className="h-6 w-6" />
                    </button>
                  </div>
                )}
                {children}
              </Dialog.Panel>
            </Transition.Child>
          </div>
        </div>
      </Dialog>
    </Transition>
  );
};
```

## 9. 响应式设计

### 9.1 断点策略
- 桌面优先设计
- Tailwind默认断点：
  - `sm`: 640px（平板竖屏）
  - `md`: 768px（平板横屏）
  - `lg`: 1024px（小桌面）
  - `xl`: 1280px（标准桌面）
  - `2xl`: 1536px（大桌面）

### 9.2 最小支持分辨率
- 1366×768

### 9.3 响应式布局示例
```tsx
<div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
  {/* 移动端1列，平板2列，桌面4列 */}
</div>

<div className="hidden lg:block">
  {/* 仅在桌面端显示 */}
</div>
```
